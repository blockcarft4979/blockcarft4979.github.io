一、架构基础：ViewModel + StateFlow

1.1 标准 ViewModel 模板

```kotlin
class XxxViewModel : ViewModel() {
    // 私有可变 StateFlow，对外暴露只读
    private val _data = MutableStateFlow<DataType?>(null)
    val data: StateFlow<DataType?> = _data.asStateFlow()

    private val _isLoading = MutableStateFlow(false)
    val isLoading: StateFlow<Boolean> = _isLoading.asStateFlow()

    private val _errorMsg = MutableStateFlow<String?>(null)
    val errorMsg: StateFlow<String?> = _errorMsg.asStateFlow()

    fun loadData() {
        viewModelScope.launch {
            _isLoading.value = true
            _errorMsg.value = null
            try {
                // 网络请求...
            } catch (e: Exception) {
                _errorMsg.value = e.message
            } finally {
                _isLoading.value = false
            }
        }
    }
}
```

1.2 Compose 中使用 StateFlow

```kotlin
// ✅ 正确写法：用 collectAsState 委托
val data by viewModel.data.collectAsState()
val isLoading by viewModel.isLoading.collectAsState()

// ❌ 错误写法：直接委托 StateFlow（编译报错）
val data by viewModel.data  // 报错！
```

二、数据持久化：TokenManager（加密存储）

2.1 TokenManager 完整实现

```kotlin
object TokenManager {
    private lateinit var appContext: Context
    private lateinit var sharedPreferences: SharedPreferences

    // ⚠️ 必须在 Application.onCreate() 中调用
    fun init(context: Context) {
        appContext = context.applicationContext
        try {
            val masterKey = MasterKey.Builder(appContext)
                .setKeyScheme(MasterKey.KeyScheme.AES256_GCM)
                .setUserAuthenticationRequired(false)  // 不强制锁屏
                .build()
            sharedPreferences = EncryptedSharedPreferences.create(
                appContext,
                "secure_prefs",
                masterKey,
                EncryptedSharedPreferences.PrefKeyEncryptionScheme.AES256_SIV,
                EncryptedSharedPreferences.PrefValueEncryptionScheme.AES256_GCM
            )
        } catch (e: Exception) {
            // 降级方案：普通 SharedPreferences
            sharedPreferences = appContext.getSharedPreferences("secure_prefs", Context.MODE_PRIVATE)
        }
    }

    fun saveAccessToken(token: String) {
        sharedPreferences.edit().putString("access_token", token).apply()
    }
    fun getAccessToken(): String? = sharedPreferences.getString("access_token", null)
    fun saveRefreshToken(token: String) = sharedPreferences.edit().putString("refresh_token", token).apply()
    fun getRefreshToken(): String? = sharedPreferences.getString("refresh_token", null)
    fun clear() = sharedPreferences.edit().clear().apply()
}
```

2.2 Application 中初始化

```kotlin
class MyApplication : Application() {
    override fun onCreate() {
        super.onCreate()
        TokenManager.init(this)  // ⚠️ 必须在任何调用之前执行

}
```

3.3 带 Token 的请求模板

```kotlin
suspend fun fetchData(): Result<DataType> {
    return withContext(Dispatchers.IO) {
        try {
            val response = NetworkClient.client.newCall(
                Request.Builder()
                    .url(API.XXX.URL)
                    .addHeader("Authorization", "Bearer ${TokenManager.getAccessToken()}")
                    .build()
            ).execute()
            
            val body = response.body?.string() ?: ""
            if (response.isSuccessful) {
                val result = JsonParser.json.decodeFromString<BaseData<DataType>>(body)
                if (result.code == "OK") Result.success(result.data)
                else Result.failure(Exception(result.message))
            } else {
                Result.failure(Exception("HTTP ${response.code}"))
            }
        } catch (e: Exception) {
            Result.failure(e)
        }
    }
}
```

3.4 Token 过期自动刷新

```kotlin
suspend fun refreshToken(): Boolean {
    val refreshToken = TokenManager.getRefreshToken()
    if (refreshToken.isNullOrEmpty()) return false
    
    val response = NetworkClient.client.newCall(
        Request.Builder()
            .url(API.Auth.REFRESH_URL)
            .post("""{"refresh_token":"$refreshToken"}""".toRequestBody())
            .build()
    ).execute()
    
    return if (response.isSuccessful) {
        val data = JsonParser.json.decodeFromString<BaseData<UserToken>>(response.body?.string() ?: "")
        data.data?.let {
            TokenManager.saveAccessToken(it.accessToken)
            TokenManager.saveRefreshToken(it.refreshToken)
            true
        } ?: false
    } else false
}
```

四、UI 层：Compose 核心

4.1 防止导航切换重复加载

问题：每次切换底部导航，LaunchedEffect(Unit) 都会重新执行，导致重复网络请求。

解决方案：在 ViewModel 中判断是否已有数据。

```kotlin
fun loadData(forceRefresh: Boolean = false) {
    if (!forceRefresh && _data.value != null) {
        return  // 已有数据，跳过加载
    }
    // ... 网络请求 ...
}
```

```kotlin
val lazyListState = LazyListState()
```

4.3 底部导航布局（避坑版）

❌ 错误：外层和内层都用 Scaffold，导致双层内边距。
✅ 正确：外层用 Scaffold，内层用 Column + TopAppBar；或外层用 Column，内层保留 Scaffold。

```kotlin
// 推荐方案：外层 Scaffold，内层用 Column
Scaffold(bottomBar = { NavBar(...) }) { innerPadding ->
    when (index) {
        0 -> HomeLayout(modifier = Modifier.padding(innerPadding))
        1 -> PostLayout(modifier = Modifier.padding(innerPadding))
    }
}

// 内层布局：接收 modifier，使用 Column + TopAppBar
@Composable
fun HomeLayout(modifier: Modifier = Modifier) {
    Column(modifier = modifier) {
        // 不需要再套 Scaffold
        LazyColumn(modifier = Modifier.weight(1f)) { ... }
    }
}
```

六、常见踩坑记录

问题 原因 解决方案
lateinit property tokenManager has not been initialized 忘记在 Application.onCreate() 中初始化，或 AndroidManifest 没注册 MyApplication 检查清单文件，确保 android:name=".MyApplication"
ExceptionInInitializerError EncryptedSharedPreferences 类加载时 appContext 未赋值 用 by lazy 延迟初始化，或在 init 中一次性赋值
头像加载失败 头像 URL 是 API 接口而非静态图片，且未带 Token 用 ImageRequest.Builder().addHeader("Authorization", ...)
切换导航重新加载 LaunchedEffect(Unit) 每次重组都执行 在 ViewModel 中判断是否已有数据
滚动位置丢失 LazyListState 随 Composable 重建 把 LazyListState 放在 ViewModel 中
双层 Scaffold 内外层都用 Scaffold，内边距叠加 外层用 Scaffold，内层用 Column + TopAppBar
Token 过期不刷新 调用 refreshToken() 但没等待结果 改为 suspend fun refreshToken(): Boolean 并等待
无网络时崩溃 execute() 抛出 IOException 未捕获 用 try-catch 包裹网络请求
